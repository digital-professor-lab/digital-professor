\begin{frame}{What got done}

      \begin{block}{A working tutoring experience}
        \begin{itemize}
          \item Browser-based course workspace

          \item Course-grounded tutoring chat

        \end{itemize}

\begin{frame}{Interaction methods now available}
  \begin{columns}[T,onlytextwidth]
    \column{0.32\textwidth}
      \begin{block}{Course material}
        Upload syllabi, notes, homework, images, and handwritten work so the tutor can answer in context.
      \end{block}
    \column{0.32\textwidth}
      \begin{block}{Live writing}
        Draw a question or equation with a mouse, finger, or stylus and submit it from the chat.
      \end{block}
    \column{0.32\textwidth}
      \begin{block}{Voice}
        Record a question, review the transcription, correct it if needed, and send it to the tutor.
      \end{block}
  \end{columns}
  \vspace{0.8em}
  \centering
  The prototype records quality feedback, correction effort, response time, and estimated cost so these methods can be compared.
\end{frame}

\begin{frame}{What we learned from the first demo}
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
      \begin{block}{Promising}
        \begin{itemize}
          \item The complete student flow is understandable
          \item Multiple input methods fit different learning tasks
          \item Mathematical output is readable
          \item Course grounding makes the tutor feel more relevant
        \end{itemize}
      \end{block}
    \column{0.49\textwidth}
      \begin{block}{Unresolved}
        \begin{itemize}
          \item We have not shown that the tutoring is consistently correct or helpful
          \item Answers can be too detailed for the student's need
          \item Recognition and citations still fail on ambiguous inputs
          \item Latency can interrupt the tutoring rhythm
        \end{itemize}
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Next milestone: mock tutoring situations}
  \begin{block}{Tune and evaluate the tutoring experience}
    Build a repeatable set of realistic student--professor interactions, score the prototype, adjust its tutoring behavior, and run the same situations again.
  \end{block}
  \vspace{0.5em}
  \begin{itemize}
    \item A student asks for a conceptual explanation at the wrong level of detail
    \item A student has a misconception and needs guided correction
    \item A student requests help on homework without being given the answer
    \item A student submits flawed work and asks for review
    \item A student asks a question that must be answered from course material
    \item A handwritten or spoken equation is ambiguous and requires clarification
  \end{itemize}
\end{frame}

\begin{frame}{Blockers and roadmap}
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
      \textbf{Current blockers}
      \begin{itemize}
        \item No agreed scoring rubric or ground-truth tutoring examples yet
        \item Handwriting and spoken mathematics remain ambiguous
        \item Response time can be too slow for natural dialogue
        \item The prototype is local and does not retain student work
      \end{itemize}
    \column{0.47\textwidth}
      \textbf{Roadmap after evaluation}
      \begin{enumerate}
        \item Improve course grounding and citations
        \item Add persistent course workspaces
        \item Pilot a secure public web application
        \item Prototype real-time Office Hours
        \item Add specialized agents only if evaluation shows a need
      \end{enumerate}
  \end{columns}
\end{frame}

\begin{frame}{Decision we want from the next milestone}
  \begin{block}{Is the prototype acting like a useful tutor?}
    The next milestone should tell us which tutoring behaviors work, where students must correct the system, and whether the experience is accurate, grounded, responsive, and affordable enough to develop further.
  \end{block}
  \vspace{0.8em}
  \begin{itemize}
    \item Keep the interaction methods that improve student success
    \item Tune explanations, hints, review behavior, and comprehension checks
    \item Use evidence to choose models and decide whether more architecture is necessary
  \end{itemize}
